# BearTracks PTO MVP — Implementation Plan

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** Build the BearTracks PTO MVP — a multi-tenant PTO app with a rule-based leave policy engine, employee self-service, and manager approvals.

**Architecture:** Pure functional policy engine (`lib/policy-engine/`) with zero DB dependencies; a DAL layer (`lib/dal/`) owning all Drizzle queries scoped by `tenantId`; thin Next.js Server Actions orchestrating DAL + engine. Nothing skips a layer; nothing imports upward.

**Tech Stack:** Next.js 15 (App Router), PostgreSQL 16, Drizzle ORM, iron-session v8, shadcn/ui, Tailwind CSS, JavaScript, Vitest, Playwright

## Global Constraints

- JavaScript only — no TypeScript files
- JSDoc `@param`/`@returns` on all engine and DAL functions
- Every DAL function takes `tenantId` as its first parameter
- Policy engine functions are pure — zero imports from `lib/dal`, `lib/db`, or Next.js
- All Server Actions return `{ ok: true, data }` or `{ ok: false, errors: [{field?, message, blocking?}] }`
- Drizzle queries never appear outside `lib/dal/`
- bcrypt cost factor: 12
- Session cookie name: `beartracks_session`
- Seed tenant slug: `acme`
- Test runner: Vitest with `globals: true`
- Integration + E2E tests use a real Postgres DB at `DATABASE_URL_TEST`

```js
/**
 * @param {number} startMonth - 1-12
 * @param {Date} date
 * @returns {{ start: Date, end: Date, year: number }}
 */
export function deriveFiscalYear(startMonth, date) {
  const month = date.getMonth() + 1
  const year = date.getFullYear()
  const fyYear = month >= startMonth ? year : year - 1
  const start = new Date(fyYear, startMonth - 1, 1)
  const end = new Date(fyYear + 1, startMonth - 1, 0)
  return { start, end, year: fyYear }
}
```

```js
/**
 * @param {Date} startDate
 * @param {Date} endDate
 * @param {number[]} workingWeek - ISO day numbers (1=Mon, 7=Sun)
 * @param {Date[]} publicHolidays
 * @param {'HALF'|null} [halfDay]
 * @returns {number}
 */
export function countWorkingDays(startDate, endDate, workingWeek, publicHolidays, halfDay = null) {
  if (halfDay === 'HALF') return 0.5

  const holidaySet = new Set(
    publicHolidays.map(d => d instanceof Date ? d.toISOString().slice(0, 10) : d)
  )

  let count = 0
  const current = new Date(startDate)
  while (current <= endDate) {
    const dow = current.getDay() === 0 ? 7 : current.getDay()
    const dateStr = current.toISOString().slice(0, 10)
    if (workingWeek.includes(dow) && !holidaySet.has(dateStr)) {
      count++
    }
    current.setDate(current.getDate() + 1)
  }
  return count
}
```

```js
/**
 * @param {object} policy
 * @param {{ start: Date, end: Date }} fiscalYear
 * @param {Date} today
 * @returns {number}
 */
export function computeBaseEntitlement(policy, fiscalYear, today) {
  if (policy.entitlement_basis === 'FIXED') {
    return Number(policy.annual_entitlement_days)
  }
  const months = monthsBetween(fiscalYear.start, today)
  const accrued = Number(policy.accrual_rate_per_month) * months
  const cap = policy.max_accrual_balance !== null && policy.max_accrual_balance !== undefined
    ? Number(policy.max_accrual_balance)
    : Infinity
  return Math.min(accrued, cap)
}

function monthsBetween(start, end) {
  const months =
    (end.getFullYear() - start.getFullYear()) * 12 +
    (end.getMonth() - start.getMonth())
  return Math.max(0, months)
}

/**
 * @param {number} base
 * @param {object} policy
 * @param {string} role
 * @returns {number}
 */
export function applyRoleOverride(base, policy, role) {
  const overrides = policy.role_overrides ?? {}
  return base + (Number(overrides[role] ?? 0))
}

/**
 * @param {number} entitlement
 * @param {object} policy
 * @param {Date} joinDate
 * @param {{ start: Date, end: Date }} fiscalYear
 * @returns {number}
 */
export function applyProration(entitlement, policy, joinDate, fiscalYear) {
  if (!policy.prorate_on_join || joinDate <= fiscalYear.start) {
    return entitlement
  }
  const totalMs = fiscalYear.end - fiscalYear.start
  const remainingMs = fiscalYear.end - joinDate
  const ratio = remainingMs / totalMs
  return roundValue(entitlement * ratio, policy.proration_rounding)
}

function roundValue(value, mode) {
  if (mode === 'CEIL') return Math.ceil(value)
  if (mode === 'HALF_UP') return Math.floor(value + 0.5)
  return Math.floor(value) // FLOOR is default
}
```

```js
export function getCarryForwardAmount(ledgerEntries, today) {
  return ledgerEntries
    .filter(e => e.entry_type === 'CARRY_FORWARD')
    .filter(e => !e.expires_on || new Date(e.expires_on) > today)
    .reduce((sum, e) => sum + Number(e.days), 0)

export function getAdjustmentsTotal(ledgerEntries) {
  return ledgerEntries
    .filter(e => e.entry_type === 'MANUAL_ADJUSTMENT')
```

```js
import { computeBaseEntitlement, applyRoleOverride, applyProration } from './entitlement.js'
import { getCarryForwardAmount, getAdjustmentsTotal } from './carry-forward.js'

/**
 * @param {{ policy: object, employee: object, fiscalYear: object, approvedRequests: object[], ledgerEntries: object[], today: Date }} inputs
 * @returns {{ available: number, used: number, total: number, carryForward: number, adjustments: number }}
 */
export function computeAvailableBalance({ policy, employee, fiscalYear, approvedRequests, ledgerEntries, today }) {
  let entitlement = computeBaseEntitlement(policy, fiscalYear, today)
  entitlement = applyRoleOverride(entitlement, policy, employee.role)
  entitlement = applyProration(entitlement, policy, new Date(employee.joinDate), fiscalYear)

  const carryForward = getCarryForwardAmount(ledgerEntries, today)
  const adjustments = getAdjustmentsTotal(ledgerEntries)
  const total = entitlement + carryForward + adjustments
  const used = approvedRequests.reduce((sum, r) => sum + Number(r.working_days), 0)

  return { available: total - used, used, total, carryForward, adjustments }
}
```

```js
/**
 * @param {{ working_days: number, start_date: string, end_date: string }} request
 * @param {{ available: number }} balance
 * @param {object} policy
 * @param {{ today: Date, blackoutPeriods: object[], teamApprovedRequests: object[], teamSize: number, minHeadcount: number }} context
 * @returns {Array<{ ok: boolean, reason?: string, blocking?: boolean }>}
 */
export function validateRequest(request, balance, policy, context) {
  return [
    checkBalance(request, balance, policy),
    checkBlackout(request, context.blackoutPeriods),
    checkNotice(request, policy, context.today),
    checkConsecutive(request, policy),
    checkHeadcount(request, context.teamApprovedRequests, context.teamSize, context.minHeadcount),
  ]
}

function checkBalance(request, balance, policy) {
  const days = Number(request.working_days)
  if (!policy.allow_negative_balance) {
    if (balance.available < days) {
      return { ok: false, reason: `Insufficient balance. Available: ${balance.available}, requested: ${days}`, blocking: true }
    }
  } else {
    const afterRequest = balance.available - days
    const limit = policy.max_negative_days !== null && policy.max_negative_days !== undefined
      ? -Number(policy.max_negative_days)
      : -Infinity
    if (afterRequest < limit) {
      return { ok: false, reason: `Would exceed maximum negative balance of ${policy.max_negative_days} days`, blocking: true }
    }
  }
  return { ok: true }
}

function checkBlackout(request, blackoutPeriods) {
  const start = new Date(request.start_date)
  const end = new Date(request.end_date)
  for (const period of blackoutPeriods) {
    if (start <= new Date(period.end) && end >= new Date(period.start)) {
      return { ok: false, reason: `Dates fall within a blackout period: ${period.label}`, blocking: true }
    }
  }
  return { ok: true }
}

function checkNotice(request, policy, today) {
  if (!policy.min_notice_days) return { ok: true }
  const minStart = new Date(today)
  minStart.setDate(minStart.getDate() + policy.min_notice_days)
  if (new Date(request.start_date) < minStart) {
    return { ok: false, reason: `Minimum ${policy.min_notice_days} days notice required`, blocking: true }
  }
  return { ok: true }
}

function checkConsecutive(request, policy) {
  const days = Number(request.working_days)
  if (policy.min_consecutive_days > 1 && days < policy.min_consecutive_days) {
    return { ok: false, reason: `Minimum ${policy.min_consecutive_days} consecutive days required`, blocking: true }
  }
  if (policy.max_consecutive_days !== null && policy.max_consecutive_days !== undefined && days > Number(policy.max_consecutive_days)) {
    return { ok: false, reason: `Maximum ${policy.max_consecutive_days} consecutive days allowed`, blocking: true }
  }
  return { ok: true }
}

function checkHeadcount(request, teamApprovedRequests, teamSize, minHeadcount) {
  const start = new Date(request.start_date)
  const end = new Date(request.end_date)
  const current = new Date(start)
  while (current <= end) {
    const out = teamApprovedRequests.filter(r =>
      new Date(r.start_date) <= current && new Date(r.end_date) >= current
    ).length
    const present = teamSize - out - 1 // -1 for the requester
    if (present < minHeadcount) {
      return { ok: false, reason: 'Team may be below minimum headcount during this period', blocking: false }
    }
    current.setDate(current.getDate() + 1)
  }
  return { ok: true }
}
```

```js
import { NextResponse } from 'next/server'

export async function middleware(request) {
  const { pathname } = request.nextUrl

  // Extract slug from first path segment
  const parts = pathname.split('/').filter(Boolean)
  const slug = parts[0]
  if (!slug) return NextResponse.next()

  // Skip auth routes and Next.js internals
  const isAuthRoute =
    pathname.includes('/login') ||
    pathname.includes('/invite/') ||
    pathname.includes('/reset-password/')

  if (!isAuthRoute) {
    const sessionCookie = request.cookies.get('beartracks_session')
    if (!sessionCookie?.value) {
      return NextResponse.redirect(new URL(`/${slug}/login`, request.url))
    }
  }

  return NextResponse.next()
}

export const config = {
  matcher: ['/((?!_next/static|_next/image|favicon\\.ico|api/).*)'],
}
```

```js
function hashToken(token) {
  return crypto.createHash('sha256').update(token).digest('hex')

export async function login(tenantSlug, email, password) {
  try {
    const [tenant] = await db.select().from(tenants).where(eq(tenants.slug, tenantSlug))
    if (!tenant) return { ok: false, errors: [{ message: 'Invalid credentials' }] }

    const user = await getUserByEmail(tenant.id, email)
    if (!user || !user.is_active) return { ok: false, errors: [{ message: 'Invalid credentials' }] }

    const valid = await bcrypt.compare(password, user.password_hash)
    if (!valid) return { ok: false, errors: [{ message: 'Invalid credentials' }] }

    const session = await getSession()
    session.userId = user.id
    session.tenantId = tenant.id
    session.role = user.role
    await session.save()

    return { ok: true, data: { role: user.role, slug: tenantSlug } }
  } catch {
    return { ok: false, errors: [{ message: 'Something went wrong' }] }
  }
}

export async function logout() {
  const session = await getSession()
  await session.destroy()
  return { ok: true }
}

export async function sendInvite(email, role) {
  try {
    const session = await getSession()
    if (!session.userId) return { ok: false, errors: [{ message: 'Unauthorized' }] }
    if (!['admin', 'manager'].includes(session.role)) {
      return { ok: false, errors: [{ message: 'Unauthorized' }] }

    const token = crypto.randomBytes(32).toString('hex')

  }
}

export async function acceptInvite(token, name, password) {
  try {
    const tokenHash = hashToken(token)
    const [invite] = await db.select().from(invites)
      .where(and(eq(invites.token_hash, tokenHash)))

    if (!invite) return { ok: false, errors: [{ message: 'Invalid or expired invite' }] }
    if (invite.accepted_at) return { ok: false, errors: [{ message: 'Invite already used' }] }
    if (invite.expires_at < new Date()) return { ok: false, errors: [{ message: 'Invite expired' }] }

      email: invite.email,
      password_hash: passwordHash,

      join_date: new Date().toISOString().slice(0, 10),

    session.userId = user.id
    session.tenantId = invite.tenant_id

  } catch {

  }
```

```js
'use server'

import { getSession } from '@/lib/session'
import { getEmployee, getDirectReports } from '@/lib/dal/employees.js'
import { getLeaveType } from '@/lib/dal/leave-types.js'
import { getPublicHolidays } from '@/lib/dal/public-holidays.js'
import {
  insertRequest, updateRequestStatus, getRequestById,
  getTeamRequestsForDateRange, getRequestsByEmployee,
} from '@/lib/dal/leave-requests.js'
import { getLedgerEntries } from '@/lib/dal/ledger.js'
import { getApprovedRequestsForFY } from '@/lib/dal/leave-requests.js'
import { computeAvailableBalance, validateRequest, countWorkingDays } from '@/lib/policy-engine/index.js'
import { deriveFiscalYear } from '@/lib/fiscal-year.js'
import { db } from '@/lib/db/index.js'
import { tenants } from '@/lib/db/schema.js'
import { eq } from 'drizzle-orm'

export async function submitLeaveRequest({ leaveTypeId, startDate, endDate, dayType, hours, note }) {
  try {
    const session = await getSession()
    const { userId: employeeId, tenantId } = session

    const [[tenant], employee, policy] = await Promise.all([
      db.select().from(tenants).where(eq(tenants.id, tenantId)),
      getEmployee(tenantId, employeeId),
      getLeaveType(tenantId, leaveTypeId),
    ])

    if (!tenant || !employee || !policy) {
      return { ok: false, errors: [{ message: 'Not found' }] }
    }

    const today = new Date()
    const fiscalYear = deriveFiscalYear(tenant.fiscal_year_start_month, today)
    const holidays = await getPublicHolidays(tenantId, today.getFullYear())

    let working_days
    if (dayType === 'HOURLY') {
      working_days = Number(hours) / tenant.hours_per_workday
    } else if (dayType === 'HALF_AM' || dayType === 'HALF_PM') {
      working_days = 0.5
    } else {
      working_days = countWorkingDays(new Date(startDate), new Date(endDate), tenant.working_week, holidays)
    }

    const [approvedRequests, ledgerEntries] = await Promise.all([
      getApprovedRequestsForFY(tenantId, employeeId, leaveTypeId, fiscalYear.start, fiscalYear.end),
      getLedgerEntries(tenantId, employeeId, leaveTypeId),
    ])

    const balance = computeAvailableBalance({
      policy, employee, fiscalYear, approvedRequests, ledgerEntries, today,
    })

    const directReports = await getDirectReports(tenantId, employee.manager_id ?? employeeId)
    const peerIds = directReports.map(r => r.id).filter(id => id !== employeeId)
    const teamRequests = await getTeamRequestsForDateRange(tenantId, peerIds, startDate, endDate)

    const request = { working_days, start_date: startDate, end_date: endDate, day_type: dayType }
    const context = {
      today,
      blackoutPeriods: [],
      teamApprovedRequests: teamRequests,
      teamSize: peerIds.length + 1,
      minHeadcount: tenant.min_headcount_threshold,
    }

    const validations = validateRequest(request, balance, policy, context)
    const blockingErrors = validations.filter(v => !v.ok && v.blocking)
    if (blockingErrors.length > 0) {
      return { ok: false, errors: blockingErrors.map(e => ({ message: e.reason, blocking: true })) }
    }

    // Re-check balance inside transaction to guard against race conditions
    const inserted = await db.transaction(async (tx) => {
      const freshRequests = await getApprovedRequestsForFY(tenantId, employeeId, leaveTypeId, fiscalYear.start, fiscalYear.end)
      const freshBalance = computeAvailableBalance({ policy, employee, fiscalYear, approvedRequests: freshRequests, ledgerEntries, today })
      const freshValidations = validateRequest(request, freshBalance, policy, context)
      const freshBlocking = freshValidations.filter(v => !v.ok && v.blocking)
      if (freshBlocking.length > 0) throw new Error('VALIDATION_FAILED')

      const [row] = await tx.insert(require('@/lib/db/schema.js').leaveRequests).values({
        tenant_id: tenantId,
        employee_id: employeeId,
        leave_type_id: leaveTypeId,
        status: policy.requires_approval ? 'pending' : 'approved',
        start_date: startDate,
        end_date: endDate,
        day_type: dayType,
        hours: hours ?? null,
        working_days: String(working_days),
        note: note ?? null,
      }).returning()
      return row
    })

    const warnings = validations.filter(v => !v.ok && !v.blocking)
    return { ok: true, data: { requestId: inserted.id, warnings } }
  } catch (err) {
    if (err.message === 'VALIDATION_FAILED') {
      return { ok: false, errors: [{ message: 'Balance changed during submission. Please try again.' }] }
    }
    return { ok: false, errors: [{ message: 'Something went wrong' }] }
  }
}

export async function approveRequest(requestId, reviewNote = '') {
  try {
    const session = await getSession()
    const { userId, tenantId, role } = session

    const req = await getRequestById(tenantId, requestId)
    if (!req) return { ok: false, errors: [{ message: 'Not found' }] }
    if (req.status !== 'pending') return { ok: false, errors: [{ message: 'Request is not pending' }] }

    if (role === 'manager') {
      const reports = await getDirectReports(tenantId, userId)
      const reportIds = reports.map(r => r.id)
      if (!reportIds.includes(req.employee_id)) {
        return { ok: false, errors: [{ message: 'Unauthorized' }] }
      }
    } else if (role !== 'admin') {
      return { ok: false, errors: [{ message: 'Unauthorized' }] }
    }

    await updateRequestStatus(tenantId, requestId, 'approved', {
      reviewed_by: userId,
      reviewed_at: new Date(),
      review_note: reviewNote,
    })

    return { ok: true }
  } catch {
    return { ok: false, errors: [{ message: 'Something went wrong' }] }
  }
}

export async function rejectRequest(requestId, reviewNote = '') {
  try {
    const session = await getSession()
    const { userId, tenantId, role } = session

    const req = await getRequestById(tenantId, requestId)
    if (!req) return { ok: false, errors: [{ message: 'Not found' }] }
    if (req.status !== 'pending') return { ok: false, errors: [{ message: 'Request is not pending' }] }

    if (role === 'manager') {
      const reports = await getDirectReports(tenantId, userId)
      if (!reports.map(r => r.id).includes(req.employee_id)) {
        return { ok: false, errors: [{ message: 'Unauthorized' }] }
      }
    } else if (role !== 'admin') {
      return { ok: false, errors: [{ message: 'Unauthorized' }] }
    }

    await updateRequestStatus(tenantId, requestId, 'rejected', {
      reviewed_by: userId,
      reviewed_at: new Date(),
      review_note: reviewNote,
    })

    return { ok: true }
  } catch {
    return { ok: false, errors: [{ message: 'Something went wrong' }] }
  }

    return { ok: true }
  } catch {
```

```js
'use client'

import { useState } from 'react'
import { useRouter } from 'next/navigation'
import { submitLeaveRequest } from '@/app/actions/leave-requests'
import { getEmployeeBalance } from '@/app/actions/balances'
import { Button } from '@/components/ui/button'
import { Card, CardContent, CardHeader, CardTitle } from '@/components/ui/card'

function countWorkingDaysClient(start, end, workingWeek, holidays) {
  const holidaySet = new Set(holidays)
  let count = 0
  const current = new Date(start)
  while (current <= new Date(end)) {
    const dow = current.getDay() === 0 ? 7 : current.getDay()
    const ds = current.toISOString().slice(0, 10)
    if (workingWeek.includes(dow) && !holidaySet.has(ds)) count++
    current.setDate(current.getDate() + 1)
  }
  return count
}

export function NewRequestForm({ slug, leaveTypes, holidays, workingWeek, employeeId }) {
  const router = useRouter()
  const [step, setStep] = useState(1)
  const [leaveTypeId, setLeaveTypeId] = useState('')
  const [startDate, setStartDate] = useState('')
  const [endDate, setEndDate] = useState('')
  const [dayType, setDayType] = useState('FULL')
  const [hours, setHours] = useState('')
  const [note, setNote] = useState('')
  const [balance, setBalance] = useState(null)
  const [workingDays, setWorkingDays] = useState(0)
  const [errors, setErrors] = useState([])
  const [warnings, setWarnings] = useState([])
  const [submitting, setSubmitting] = useState(false)

  const selectedType = leaveTypes.find(lt => lt.id === leaveTypeId)

  async function fetchBalance(ltId, sd, ed) {
    if (!ltId || !sd || !ed) return
    const days = dayType === 'HALF_AM' || dayType === 'HALF_PM'
      ? 0.5
      : countWorkingDaysClient(sd, ed, workingWeek, holidays)
    setWorkingDays(days)
    const res = await getEmployeeBalance(employeeId, ltId)
    if (res.ok) setBalance(res.data)
  }

  async function handleSubmit() {
    setSubmitting(true)
    setErrors([])
    const result = await submitLeaveRequest({ leaveTypeId, startDate, endDate, dayType, hours: hours || null, note })
    setSubmitting(false)
    if (!result.ok) { setErrors(result.errors); return }
    setWarnings(result.data.warnings ?? [])
    router.push(`/${slug}/dashboard`)
  }

  return (
    <div className="max-w-lg space-y-6">
      <h1 className="text-2xl font-semibold">New Leave Request</h1>

      {/* Step indicators */}
      <div className="flex gap-2">
        {[1,2,3,4,5].map(s => (
          <div key={s} className={`h-1.5 flex-1 rounded-full ${s <= step ? 'bg-blue-500' : 'bg-gray-200'}`} />
        ))}
      </div>

      <Card>
        <CardContent className="pt-6 space-y-4">

          {step === 1 && (
            <div className="space-y-3">
              <p className="font-medium">Select leave type</p>
              {leaveTypes.map(lt => (
                <button
                  key={lt.id}
                  onClick={() => { setLeaveTypeId(lt.id); fetchBalance(lt.id, startDate, endDate) }}
                  className={`w-full text-left p-3 rounded-lg border flex items-center gap-3 ${leaveTypeId === lt.id ? 'border-blue-500 bg-blue-50' : 'border-gray-200'}`}
                >
                  <span className="w-3 h-3 rounded-full flex-shrink-0" style={{ background: lt.color }} />
                  <span className="text-sm font-medium">{lt.name}</span>
                </button>
              ))}
            </div>
          )}

          {step === 2 && (
            <div className="space-y-3">
              <p className="font-medium">Select dates</p>
              <div>
                <label className="text-sm text-gray-600">Start date</label>
                <input
                  type="date"
                  value={startDate}
                  onChange={e => { setStartDate(e.target.value); fetchBalance(leaveTypeId, e.target.value, endDate) }}
                  className="w-full mt-1 border rounded-md px-3 py-2 text-sm"
                />
              </div>
              <div>
                <label className="text-sm text-gray-600">End date</label>
                <input
                  type="date"
                  value={endDate}
                  min={startDate}
                  onChange={e => { setEndDate(e.target.value); fetchBalance(leaveTypeId, startDate, e.target.value) }}
                  className="w-full mt-1 border rounded-md px-3 py-2 text-sm"
                />
              </div>
              {workingDays > 0 && (
                <p className="text-sm text-gray-600">{workingDays} working day{workingDays !== 1 ? 's' : ''}</p>
              )}
            </div>
          )}

          {step === 3 && (
            <div className="space-y-3">
              <p className="font-medium">Day type</p>
              {['FULL', ...(selectedType?.allow_half_day ? ['HALF_AM', 'HALF_PM'] : []), ...(selectedType?.allow_hourly ? ['HOURLY'] : [])].map(dt => (
                <button
                  key={dt}
                  onClick={() => setDayType(dt)}
                  className={`w-full text-left p-3 rounded-lg border text-sm ${dayType === dt ? 'border-blue-500 bg-blue-50' : 'border-gray-200'}`}
                >
                  {{ FULL: 'Full day', HALF_AM: 'Half day (AM)', HALF_PM: 'Half day (PM)', HOURLY: 'Specific hours' }[dt]}
                </button>
              ))}

                <input
                  type="number"
                  placeholder="Hours"
                  value={hours}
                  onChange={e => setHours(e.target.value)}

            </div>

          <Button onClick={handleSubmit} disabled={submitting}>
            {submitting ? 'Submitting...' : 'Submit request'}
          </Button>
        )}
      </div>
    </div>
  )
}
```

```js
'use client'

import { useState } from 'react'
import { useRouter } from 'next/navigation'
import { approveRequest, rejectRequest } from '@/app/actions/leave-requests'
import { Button } from '@/components/ui/button'

export function ApprovalActions({ requestId }) {
  const router = useRouter()
  const [rejecting, setRejecting] = useState(false)
  const [note, setNote] = useState('')
  const [loading, setLoading] = useState(false)

  async function handleApprove() {
    setLoading(true)
    await approveRequest(requestId)
    router.refresh()
  }

  async function handleReject() {
    setLoading(true)
    await rejectRequest(requestId, note)
    router.refresh()
  }

  if (rejecting) {
    return (
      <div className="flex items-center gap-2">
        <input
          autoFocus
          value={note}
          onChange={e => setNote(e.target.value)}
          placeholder="Reason (optional)"
          className="border rounded px-2 py-1 text-sm w-40"
        />
        <Button size="sm" variant="destructive" onClick={handleReject} disabled={loading}>
          Confirm
        </Button>
        <Button size="sm" variant="ghost" onClick={() => setRejecting(false)}>
          Cancel
        </Button>
      </div>
    )
  }

  return (
    <div className="flex gap-2">
      <Button size="sm" onClick={handleApprove} disabled={loading}>Approve</Button>
      <Button size="sm" variant="outline" onClick={() => setRejecting(true)} disabled={loading}>
        Reject
      </Button>
    </div>
  )
}
```

**Interfaces:**

- Produces: POST endpoint that processes FY rollover for all eligible tenants

- [ ] **Step 1: Create `app/api/cron/fy-rollover/route.js`**

```js
import { db } from '@/lib/db/index.js'
import { tenants, users, leaveTypes } from '@/lib/db/schema.js'
import { eq, and } from 'drizzle-orm'
import { getApprovedRequestsForFY } from '@/lib/dal/leave-requests.js'
import { getLedgerEntries, insertLedgerEntry } from '@/lib/dal/ledger.js'
import { getEmployee } from '@/lib/dal/employees.js'
import { computeAvailableBalance } from '@/lib/policy-engine/index.js'
import { deriveFiscalYear } from '@/lib/fiscal-year.js'
import { toZonedTime, format } from 'date-fns-tz'

export async function POST(request) {
  const authHeader = request.headers.get('authorization')
  if (authHeader !== `Bearer ${process.env.CRON_SECRET}`) {
    return Response.json({ error: 'Unauthorized' }, { status: 401 })
  }

  const allTenants = await db.select().from(tenants)
  let processed = 0

  for (const tenant of allTenants) {
    // Compare today in tenant's timezone
    const nowInTz = toZonedTime(new Date(), tenant.timezone)
    const todayStr = format(nowInTz, 'MM-dd', { timeZone: tenant.timezone })
    const fyStartStr = String(tenant.fiscal_year_start_month).padStart(2, '0') + '-01'

    if (todayStr !== fyStartStr) continue

    const today = new Date()
    const expiringFY = deriveFiscalYear(tenant.fiscal_year_start_month, new Date(today.getFullYear(), today.getMonth() - 1, 15))
    const newFY = deriveFiscalYear(tenant.fiscal_year_start_month, today)

    const [allUsers, allLeaveTypes] = await Promise.all([
      db.select().from(users).where(and(eq(users.tenant_id, tenant.id), eq(users.is_active, true))),
      db.select().from(leaveTypes).where(and(eq(leaveTypes.tenant_id, tenant.id), eq(leaveTypes.allow_carry_forward, true), eq(leaveTypes.active, true))),
    ])

    for (const user of allUsers) {
      for (const lt of allLeaveTypes) {
        const [approvedRequests, ledgerEntries] = await Promise.all([
          getApprovedRequestsForFY(tenant.id, user.id, lt.id, expiringFY.start, expiringFY.end),
          getLedgerEntries(tenant.id, user.id, lt.id),
        ])

        const employee = await getEmployee(tenant.id, user.id)
        const closingBalance = computeAvailableBalance({
          policy: lt, employee, fiscalYear: expiringFY,
          approvedRequests, ledgerEntries, today: expiringFY.end,
        })

        if (closingBalance.available <= 0) continue

        const maxCarry = lt.carry_forward_max_days !== null ? Number(lt.carry_forward_max_days) : Infinity
        const carryAmount = Math.min(closingBalance.available, maxCarry)
        const forfeited = closingBalance.available - carryAmount

        let expiresOn = null
        if (lt.carry_forward_expiry_days) {
          const expiry = new Date(newFY.start)
          expiry.setDate(expiry.getDate() + lt.carry_forward_expiry_days)
          expiresOn = expiry.toISOString().slice(0, 10)
        }

        await insertLedgerEntry(tenant.id, {
          employee_id: user.id,
          leave_type_id: lt.id,
          entry_type: 'CARRY_FORWARD',
          days: String(carryAmount),
          fiscal_year: newFY.year,
          expires_on: expiresOn,
          reason: `Carry-forward from FY${expiringFY.year}`,
        })

        if (forfeited > 0) {
          console.log(`[FY ROLLOVER] ${user.email} forfeited ${forfeited} days of ${lt.code} from FY${expiringFY.year}`)
        }

        processed++
      }
    }
  }

  return Response.json({ ok: true, processed })
}
```
